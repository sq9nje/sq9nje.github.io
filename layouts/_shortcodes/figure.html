{{- /* Overrides Hugo's built-in figure so the image goes through the same WebP ladder as everything else and carries intrinsic dimensions. The render hook cannot produce a caption, which is the only reason this exists. {{< figure src="shot.png" alt="…" caption="…" >}} */ -}} {{- $src := .Get "src" }} {{- $t := partialCached "tokens.html" .Page "tokens" }} {{- $img := "" }} {{- with $.Page.Resources.GetMatch $src }}{{ $img = . }}{{ end }} {{- if not $img }}{{ with resources.Get $src }}{{ $img = . }}{{ end }}{{ end }} {{- /* Site override: without an explicit link, the image links to its full-size original and is picked up by the lightbox (assets/js/lightbox.js). Without JS the link simply opens the image. */}} {{- $full := "" }} {{- if not (.Get "link") }}{{ $full = cond (and $img (eq $img.ResourceType "image")) $img.RelPermalink $src }}{{ end }}
